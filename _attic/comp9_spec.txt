Purpose
The Intelligence Lab (Flask, localhost:5002) and the Pipeline Interpreter consume
different data sources, producing different ticker universes. Tickers appearing in
the Interpreter but absent from the Lab export have no Lab-side validation and
must be flagged for manual review before any capital decision.
This component adds a reconciliation gate that runs BEFORE /triage, compares the
two universes, surfaces mismatches, and injects Lab field data into /ticker and
/story outputs so both systems are always in view simultaneously.
1. Overview
1.1 The Problem
The Pipeline Interpreter consumes more files than the Intelligence Lab and therefore surfaces a broader ticker universe. When a ticker appears in the Interpreter but returns nil in the Lab, the trader has no way of knowing whether that ticker is genuinely valid or whether the Lab has filtered it out for a structural reason. Acting on an Interpreter verdict for a Lab-absent ticker is the risk this component eliminates.
1.2 The Solution
Manual Lab export saved to MA_Inputs/lab_export/, auto-detected by the Interpreter, reconciled before /triage fires. Three outputs:
Confirmed — ticker in both. Full Lab field context injected into /ticker and /story.
Lab only — ticker ranked by Lab but not surfaced by Interpreter. Shown as missed candidates.
Interpreter only — ticker in Interpreter but NOT in Lab. Carries LAB_NOT_CONFIRMED warning on every verdict. Flagged for manual review.
1.3 Isolation Guarantee
This component cannot affect the AVSHUNTER pipeline
lab_reconciliation.py has zero dependency on intelligent_orchestrator.py,
morning_thesis_validator.py, or any pipeline run file.
It reads one CSV from MA_Inputs/lab_export/ and compares it to in-memory
pipeline rows already loaded by the Interpreter. Pure comparison logic only.

3.2 load_lab_export()
def load_lab_export(ma_lab_dir: Path) -> tuple[list[dict], str]:
"""
Find and load the most recent avshunter_signals_*.csv
from MA_Inputs/lab_export/.
Normalises Ticker column to uppercase on load.
Returns (rows, filename). Returns ([], "") if none found.
"""
if not ma_lab_dir.exists():
return [], ""
candidates = sorted(
ma_lab_dir.glob("avshunter_signals_*.csv"),
key=lambda p: p.stat().st_mtime,
reverse=True,
)
if not candidates:
return [], ""
path = candidates[0]
rows = []
with open(path, "r", encoding="utf-8-sig") as f:
for r in csv.DictReader(f):
r["Ticker"] = str(r.get("Ticker", "")).strip().upper()
rows.append(r)
return rows, path.name
3.3 reconcile_universes()
def reconcile_universes(
lab_rows: list[dict],
interpreter_tickers: list[str],
) -> dict:
"""
Three-way comparison between Lab export and Interpreter universe.
interpreter_tickers: list of uppercase ticker strings from pipeline rows.
Returns dict with all three lists plus summary counts.
"""
lab_set   = {r["Ticker"] for r in lab_rows if r.get("Ticker")}
interp_set = {t.upper().strip() for t in interpreter_tickers if t}
confirmed   = sorted(lab_set & interp_set)
lab_only    = sorted(lab_set - interp_set)
interp_only = sorted(interp_set - lab_set)
return {
"confirmed":        confirmed,
"lab_only":         lab_only,
"interp_only":      interp_only,
"confirmed_count":  len(confirmed),
"lab_only_count":   len(lab_only),
"interp_only_count":len(interp_only),
}
3.4 get_lab_row()
def get_lab_row(lab_rows: list[dict], ticker: str) -> dict | None:
"""
Return the Lab row for a specific ticker, or None.
Called by /ticker and /story to inject Lab context into prompts.
"""
ticker = ticker.upper().strip()
for r in lab_rows:
if r.get("Ticker") == ticker:
return r
return None
3.5 check_run_id_alignment()
def check_run_id_alignment(
lab_rows: list[dict],
pipeline_run_id: str,
) -> dict:
"""
Verify the Lab export Run_ID matches the pipeline run being interpreted.
Lab and pipeline must share the same Run_ID for reconciliation to be valid.
Returns dict: { match: bool, lab_run_id: str, pipeline_run_id: str }
"""
if not lab_rows:
return { "match": False, "lab_run_id": "UNKNOWN", "pipeline_run_id": pipeline_run_id }
lab_run_id = str(lab_rows[0].get("Run_ID", "")).strip()
match = lab_run_id == str(pipeline_run_id).strip()
return { "match": match, "lab_run_id": lab_run_id, "pipeline_run_id": pipeline_run_id }
3.6 validate_lab_field_alignment()
def validate_lab_field_alignment(
lab_row: dict,
pipeline_row: dict,
ticker: str,
) -> list[dict]:
"""
Field-level comparison for a single ticker.
Returns list of conflict dicts — empty list means fully aligned.
Each conflict: { field, lab_value, pipeline_value, severity }
"""
conflicts = []
for lab_field, pipe_field in FIELD_MAP.items():
lab_val  = str(lab_row.get(lab_field, "")).strip()
pipe_val = str(pipeline_row.get(pipe_field, "")).strip()
if not lab_val or not pipe_val:
continue  # skip if either side is empty
# Numeric tolerance check for IVP and Premium
if lab_field in ("IVP", "Premium_Mid"):
try:
lv, pv = float(lab_val), float(pipe_val)
tol = IVP_TOLERANCE if lab_field == "IVP" else (pv * PREMIUM_TOLERANCE)
if abs(lv - pv) <= tol:
continue
except ValueError:
pass
# Direction normalisation — PUT/LONG_PUT both mean same direction
if lab_field == "Direction":
lab_norm  = "PUT" if "PUT" in lab_val.upper()  else ("CALL" if "CALL" in lab_val.upper() else lab_val)
pipe_norm = "PUT" if "PUT" in pipe_val.upper() else ("CALL" if "CALL" in pipe_val.upper() else pipe_val)
if lab_norm == pipe_norm:
continue
if lab_val.upper() != pipe_val.upper():
conflicts.append({
"field":          lab_field,
"lab_value":      lab_val,
"pipeline_value": pipe_val,
"severity":       FIELD_SEVERITY.get(lab_field, "WARN"),
})
return conflicts
3.7 build_lab_alignment_block()
def build_lab_alignment_block(
reconciliation: dict,
lab_filename: str,
run_id_check: dict = None,
) -> str:
"""
Build LAB_ALIGNMENT text block for injection into triage prompt.
"""
lines = [f"LAB_ALIGNMENT ({lab_filename}):"]
if run_id_check:
status = "MATCH" if run_id_check["match"] else "MISMATCH — STALE_LAB_DATA WARNING"
lines.append(f"  run_id_check: {status}")
lines.append(f"    lab_run_id:      {run_id_check['lab_run_id']}")
lines.append(f"    pipeline_run_id: {run_id_check['pipeline_run_id']}")
lines.append(f"  confirmed:    {reconciliation['confirmed_count']} tickers in both Lab and Interpreter")
lines.append(f"  lab_only:     {reconciliation['lab_only_count']} tickers in Lab not surfaced by Interpreter")
lines.append(f"  interp_only:  {reconciliation['interp_only_count']} tickers — REVIEW REQUIRED")
if reconciliation["interp_only"]:
lines.append(f"    {', '.join(reconciliation['interp_only'])}")
return "\n".join(lines)
3.8 build_lab_context_block()
def build_lab_context_block(lab_row: dict, ticker: str) -> str:
"""
Build a LAB_CONTEXT block for injection into /ticker and /story prompts.
Surfaces the key Lab fields alongside the pipeline row.
"""
if not lab_row:
return f"LAB_CONTEXT_{ticker}: NOT_IN_LAB_EXPORT — LAB_NOT_CONFIRMED"
fields = [
"Verdict", "Lab_Verdict", "Direction", "Conv_Score",
"Strike", "Premium_Mid", "RR", "EV", "IVP", "IVP_Label",
"Priority_Rank", "Priority_Score", "WBS_Grade", "EIL_Verdict",
"Exec_Category", "Vol_State", "Run_ID",
]
lines = [f"LAB_CONTEXT_{ticker}:"]
for f in fields:
v = lab_row.get(f, "")
if str(v).strip() not in ("", "nan", "None"):
lines.append(f"  {f}: {v}")
return "\n".join(lines)
3.9 build_field_conflict_block()
def build_field_conflict_block(conflicts: list[dict], ticker: str) -> str:
"""
Format field-level conflicts for injection into /ticker and /story prompts.
"""
if not conflicts:
return f"LAB_FIELD_CONFLICTS_{ticker}: NONE — fully aligned"
lines = [f"LAB_FIELD_CONFLICTS_{ticker} ({len(conflicts)} conflicts):"]
for c in conflicts:
lines.append(
f"  {c['field']:<16} | Lab: {c['lab_value']:<20} | Pipeline: {c['pipeline_value']:<20} | {c['severity']}"
)
flag_count = sum(1 for c in conflicts if c["severity"] == "FLAG")
if flag_count:
lines.append(f"  {flag_count} FLAG-level conflict(s) — surface in RISK ASSESSMENT")
return "\n".join(lines)

def cmd_lab(args: str = ""):
"""
/lab [FILE]
Manually load a Lab export CSV and show reconciliation summary.
If FILE not specified, scans MA_Inputs/lab_export/ for most recent file.
Does not run a full triage -- reconciliation summary only.
"""
from lab_reconciliation import (
load_lab_export, reconcile_universes,
check_run_id_alignment, build_lab_alignment_block,
)
path = args.strip().strip('"').strip("'") if args.strip() else None
if path:
from pathlib import Path as _P
lab_rows, lab_filename = load_lab_export(_P(path).parent)
else:
lab_rows, lab_filename = load_lab_export(MA_LAB)
if not lab_rows:
print("  ⚠  No lab export found.")
print(f"  Place avshunter_signals_*.csv in {MA_LAB}")
return
interpreter_tickers = []
if SESSION.get("pipeline_csv"):
from pipeline_interpreter_engine import read_pipeline_csv
rows = read_pipeline_csv(SESSION["pipeline_csv"])
interpreter_tickers = [str(r.get("ticker","")).upper() for r in rows]

reconciliation  = reconcile_universes(lab_rows, interpreter_tickers)
lab_block       = build_lab_alignment_block(reconciliation, lab_filename, run_id_check)
print(f"\n{lab_block}")
SESSION["lab_rows"]           = lab_rows
SESSION["lab_reconciliation"] = reconciliation
SESSION["lab_filename"]       = lab_filename
print(f"\n  Lab export loaded into session. Run /triage to use reconciliation.")

INTELLIGENCE LAB CONFLICT RULES
═══════════════════════════════════════════════
When a LAB_CONTEXT_{TICKER} block is present in the prompt:
1. Read it before producing any analysis for that ticker.
2. Surface the Lab verdict, Lab direction, and Lab priority rank

- Strike conflict    → note in EXECUTION PRESCRIPTION
3. For WARN-severity conflicts: note in EVIDENCE QUALITY, do not escalate.
4. Never block a verdict solely due to Lab conflicts — surface and note only.

1. Add LAB_NOT_CONFIRMED to the RISK ASSESSMENT conflict list.
2. In FINAL VERDICT add: "Note: ticker not present in Intelligence Lab export
— confirm Lab-side before execution decision."
3. Do not block the verdict — flag only.
When LAB_ALIGNMENT block is present in a triage prompt:
1. Include the reconciliation summary in SESSION_SUMMARY.
2. Flag every interp_only ticker in the triage ranked table.
3. If run_id_check shows MISMATCH: add STALE_LAB_DATA warning to

def check_lab_reconciliation_qa(
lab_rows: list,
lab_filename: str,
reconciliation: dict,
run_id_check: dict,
) -> dict:
"""
QA checks for the lab reconciliation layer.
Called after lab export is loaded in cmd_triage().
"""
results = { "mode": "LAB_RECONCILIATION", "checks": {} }
# 1. Lab file loaded
results["checks"]["lab_file_loaded"] = {
"status": "PASS" if lab_rows else "WARN",
"filename": lab_filename or "NOT_FOUND",
"rows": len(lab_rows),
}
# 2. Run_ID alignment
if run_id_check:
results["checks"]["run_id_alignment"] = {
"status": "PASS" if run_id_check["match"] else "WARN",
"lab_run_id": run_id_check["lab_run_id"],
"pipeline_run_id": run_id_check["pipeline_run_id"],
"note": "" if run_id_check["match"] else "STALE_LAB_DATA — different sessions",
}
# 3. Reconciliation completeness
if reconciliation:
interp_only_count = reconciliation.get("interp_only_count", 0)
results["checks"]["reconciliation"] = {
"status": "PASS" if interp_only_count == 0 else "WARN",
"confirmed": reconciliation.get("confirmed_count", 0),
"lab_only": reconciliation.get("lab_only_count", 0),
"interp_only": interp_only_count,
"note": f"{interp_only_count} tickers require manual review" if interp_only_count else "",
}
return results
